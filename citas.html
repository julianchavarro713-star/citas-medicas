function descargarPDF(idCita) { fetch(`${API_URL}/citas`) .then(res => res.json()) .then(citas => { const cita = citas.find(c => c._id === idCita); if (!cita) { alert("❌ No se encontró la cita"); return; } // Obtener teléfono del usuario actual const usuarioActual = JSON.parse(localStorage.getItem("usuario")); const telefonoUsuario = usuarioActual.telefono || "No registrado"; const fechaFormateada = cita.fecha.split('T')[0]; const fechaEmision = new Date().toLocaleString(); const idCorta = idCita.substring(0, 8).toUpperCase(); const motivoTexto = (cita.motivo && cita.motivo !== "No especificado") ? cita.motivo : "Control y Seguimiento"; // Direcciones por sede const direccionesSedes = { "Urgencias Toberin": "Calle 45 #20-30, Toberín", "La Cardio": "Avenida 19 #120-45, Chapinero", "Colina": "Carrera 7 #180-50, Usaquén" }; // Consultorio por sede const consultorio = cita.centroMedico === "Urgencias Toberin" ? "Consultorio 101" : cita.centroMedico === "La Cardio" ? "Consultorio 205" : "Consultorio 308"; // Obtener especialidad del doctor let especialidad = ""; for (let sede in doctoresPorSede) { const doctor = doctoresPorSede[sede].find(d => d.nombre === cita.doctor); if (doctor) { especialidad = doctor.especialidad; break; } } const estado = "CONFIRMADA"; const contenido = ` Comprobante de Cita

🏥 MediCitas Pro

COMPROBANTE DE CITA MÉDICA

${fechaEmision}

📋 DATOS DE LA CITA
Número de cita:
${idCorta}
Paciente:
${cita.paciente}
Teléfono:
${telefonoUsuario}
Fecha:
${fechaFormateada}
Hora:
${cita.hora}
Estado:
${estado}
🏥 DATOS DEL CENTRO MÉDICO
Centro:
${cita.centroMedico}
Dirección:
${direccionesSedes[cita.centroMedico] || "Dirección no disponible"}
Consultorio:
${consultorio}
👨‍⚕️ DATOS DEL ESPECIALISTA
Doctor:
${cita.doctor}
Especialidad:
${especialidad || "Médico General"}
Tipo de cita:
${motivoTexto}
Gracias por confiar en MediCitas Pro
www.medicitaspro.com
`; // Crear elemento y generar PDF const element = document.createElement('div'); element.innerHTML = contenido; document.body.appendChild(element); html2pdf().from(element).set({ margin: 0.5, filename: `Cita_${cita.paciente}_${fechaFormateada}.pdf`, html2canvas: { scale: 2 }, jsPDF: { unit: 'in', format: 'letter', orientation: 'portrait' } }).save().then(() => { document.body.removeChild(element); }).catch(err => { console.error("Error:", err); alert("❌ Error al generar el PDF"); document.body.removeChild(element); }); }); }
